import math

// Inputs
input point : Float64
input state : Float64
input gesture : Float64
input geofence : Float64
input proxy : Float64
input vel : Float64

// AtomicPropositions
output ap12 : Float64 := (gesture - 3.00)
output ap6 : Float64 := (gesture - 0.00)
output ap0 : Float64 := geofence
output ap10 : Float64 := (vel - 1.00)
output ap14 : Float64 := (state - proxy)
output ap2 : Float64 := (gesture - 1.00)
output ap16 : Float64 := (vel - 1.00)

// Leaf transitions (represent which leaf node is active)
input leaf_0 : Bool
input leaf_1 : Bool

input leaf_4 : Bool
input leaf_5 : Bool
input leaf_8 : Bool
input leaf_9 : Bool

// Node transitions (checks if leaf activation is valid wrt TBT)
output node_transition_0: Bool := !leaf_0.offset(by:-1).defaults(to: false) or leaf_0
output node_transition_1: Bool := !leaf_1.offset(by:-1).defaults(to: false) or (leaf_2 or leaf_1)
output node_transition_2: Bool := !leaf_2.offset(by:-1).defaults(to: false) or (leaf_3 or leaf_8 or leaf_2)
output node_transition_3: Bool := !leaf_3.offset(by:-1).defaults(to: false) or (leaf_4 or leaf_5 or leaf_3)
output node_transition_4: Bool := !leaf_4.offset(by:-1).defaults(to: false) or (leaf_15 or leaf_4)
output node_transition_5: Bool := !leaf_5.offset(by:-1).defaults(to: false) or (leaf_15 or leaf_5)
output node_transition_8: Bool := !leaf_8.offset(by:-1).defaults(to: false) or (leaf_9 or leaf_10 or leaf_8)
output node_transition_9: Bool := !leaf_9.offset(by:-1).defaults(to: false) or (leaf_15 or leaf_9)
output node_transition_10: Bool := !leaf_10.offset(by:-1).defaults(to: false) or (leaf_15 or leaf_10)
output node_transition_15: Bool := !leaf_15.offset(by:-1).defaults(to: false) or (leaf_3 or leaf_8 or leaf_15)
output valid_node_transitions : Bool := node_transition_0 and node_transition_1 and node_transition_2 and node_transition_3 and node_transition_4 and node_transition_5 and node_transition_8 and node_transition_9 and node_transition_10 and node_transition_15 

// TL monitors
//- tl_transtion maps from one state to the next
//- tl_outcome is the labeling fnc
//- triggers are only produces once another leaf activates
// G(ap0)
output tl_transition_0 : UInt
  spawn when leaf_0
  eval with if tl_transition_0.offset(by: -1).defaults(to: 1) == 1 and (ap0<0.0) then 0 else
            if tl_transition_0.offset(by: -1).defaults(to: 1) == 1 and (ap0>0.0) then 1 else
            if tl_transition_0.offset(by: -1).defaults(to: 1) == 0 and (ap0>0.0) then 0 else
               tl_transition_0.offset(by: -1).defaults(to: 1)
  close when !leaf_0

output tl_outcome_0 : Int
  spawn when leaf_0
  eval with if tl_transition_0.offset(by: -1).defaults(to: 1) == 1 and tl_transition_0 == 0 then -2 else
            if tl_transition_0.offset(by: -1).defaults(to: 1) == 1 and tl_transition_0 == 1 then 1 else
            if tl_transition_0.offset(by: -1).defaults(to: 1) == 0 and tl_transition_0 == 0 then -2 else
               tl_outcome_0.offset(by: -1).defaults(to: 1)
  close when !leaf_0

output tl_change_leaf_0
  spawn when leaf_0
  eval with tl_outcome_0.offset(by: -1).defaults(to: 0) != tl_outcome_0
  close when !leaf_0

trigger spawn when leaf_0
  eval when tl_change_leaf_0 and tl_outcome_0 == 2 with "Satisfied Leaf node 0"
  close when !leaf_0

trigger spawn when leaf_0
  eval when tl_change_leaf_0 and tl_outcome_0 == 1 with "Presumably satisfied Leaf node 0"
  close when !leaf_0

trigger spawn when leaf_0
  eval when tl_change_leaf_0 and tl_outcome_0 == -1 with "Presumably violated Leaf node 0"
  close when !leaf_0

trigger spawn when leaf_0
  eval when tl_change_leaf_0 and tl_outcome_0 == -2 with "Violated Leaf node 0"
  close when !leaf_0
